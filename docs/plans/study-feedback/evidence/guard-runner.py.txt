import os,sys,json,time,subprocess,signal,tarfile,shutil
from pathlib import Path
WORKER='edict-validation'
CACHE='edict-build-cache'
ROOT=Path('/Users/j/git/edict-study-feedback')
SCRATCH=Path('/Users/j/.codex/tmp/edict-feedback')
GiB=1024**3

def run(args,**kw):
    return subprocess.run(args,check=True,timeout=kw.pop('timeout',30),**kw)
def read(args):
    return run(args,stdout=subprocess.PIPE,text=True).stdout.strip()
def stop():
    subprocess.run(['docker','stop','--timeout','5',WORKER],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,timeout=15)
def metrics():
    stat=json.loads(read(['docker','inspect','--size',WORKER]))[0]
    host=shutil.disk_usage(SCRATCH).free
    out=read(['docker','exec',WORKER,'sh','-c','du -sk /cache /tmp /work /data; df -Pk /cache'])
    lines=out.splitlines(); sizes=[int(line.split()[0])*1024 for line in lines[:4]]
    vm=int(lines[-1].split()[3])*1024
    logs=sum(p.stat().st_size for p in SCRATCH.glob('*.log'))
    return dict(build=sum(sizes),data=sum(sizes[1:])+stat.get('SizeRw',0)+sum(p.stat().st_size for p in SCRATCH.glob('*.tar'))+sum(p.stat().st_size for p in SCRATCH.glob('*.bundle')),logs=logs,layer=stat.get('SizeRw',0),host_free=host,vm_free=vm)
def guard(m):
    assert m['host_free'] >= 50*GiB and m['vm_free'] >= 50*GiB, 'free space below 50 GiB'
    assert m['build'] < 20*GiB, 'project build cache reached 20 GiB'
    assert m['data'] < 4*GiB-128*1024**2, 'project data budget reached'
    assert m['logs'] < 124*1024**2, 'host logs reached budget (4 MiB reserved for Docker logs)'
    assert m['layer'] == 237568, 'readonly container layer changed from inspected OrbStack certificate baseline'

def prepare():
    known=run(['docker','container','ls','-a','--filter',f'name=^{WORKER}$','--format','{{.Names}}'],stdout=subprocess.PIPE,text=True).stdout.strip()
    if not known:
        run(['docker','volume','create',CACHE],stdout=subprocess.DEVNULL)
        run(['docker','create','--name',WORKER,'--label','project=edict','--read-only','--cpus','4','--memory','12g','--memory-swap','12g','--pids-limit','512','--log-driver','json-file','--log-opt','max-size=1m','--log-opt','max-file=3','--mount',f'type=volume,src={CACHE},dst=/cache','--tmpfs','/work:rw,exec,size=128m','--tmpfs','/tmp:rw,exec,size=512m','--tmpfs','/data:rw,exec,size=3g','--env','CARGO_HOME=/cache/cargo','--env','CARGO_TARGET_DIR=/cache/target','--env','CARGO_INCREMENTAL=0','--env','CARGO_BUILD_JOBS=4','--env','CARGO_PROFILE_DEV_DEBUG=0','--env','CARGO_PROFILE_TEST_DEBUG=0','--env','TMPDIR=/data','rust:1.96.0','sleep','infinity'],stdout=subprocess.DEVNULL)
    cfg=json.loads(read(['docker','inspect',WORKER]))[0]
    assert cfg['Config']['Labels'].get('project') == 'edict'
    assert cfg['HostConfig']['ReadonlyRootfs'] and cfg['HostConfig']['Memory'] == 12*GiB
    assert cfg['Mounts'][0]['Name']==CACHE and len(cfg['Mounts'])==1
    run(['docker','start',WORKER],stdout=subprocess.DEVNULL)
    m=metrics();guard(m)
    print(json.dumps({'launch_contract':{'worker':WORKER,'image':'rust:1.96.0','lock':['host/heavy-work','host/docker/edict-validation/'],'outputs':['/cache cargo+target+owned toolchain','/work source+git snapshot','/tmp','/data tests',str(SCRATCH)+' bounded logs+snapshots'],'build_limit':20*GiB,'data_limit':4*GiB,'log_limit':128*1024**2,'cpu':4,'memory':12*GiB,'guard':'fail-closed 2-second usage monitor; read-only layer; bounded tmpfs; 3 x 1MiB Docker logs','timeout_seconds':2400,'stop':WORKER,'measurements':m}}),flush=True)
    run(['git','bundle','create',str(SCRATCH/'source.bundle'),'--all'],cwd=ROOT)
    files=read(['git','ls-files','--cached','--others','--exclude-standard']).splitlines() if Path.cwd()==ROOT else read(['git','-C',str(ROOT),'ls-files','--cached','--others','--exclude-standard']).splitlines()
    if os.environ.get('EDICT_PARENT_WITNESS') == '1':
        files=['crates/edict-syntax/tests/return_budget.rs']
    with tarfile.open(SCRATCH/'source.tar','w') as archive:
        for file in files: archive.add(ROOT/file,arcname=file,recursive=False)
    guard(metrics())
    run(['docker','exec',WORKER,'sh','-c','rm -rf /work/edict; mkdir -p /work/edict'])
    with (SCRATCH/'source.bundle').open('rb') as bundle:
        run(['docker','exec','-i',WORKER,'sh','-c','cat > /work/source.bundle'],stdin=bundle)
    run(['docker','exec',WORKER,'git','clone','--quiet','/work/source.bundle','/work/edict'])
    branch=read(['git','-C',str(ROOT),'branch','--show-current'])
    run(['docker','exec',WORKER,'git','-C','/work/edict','checkout',branch],stdout=subprocess.DEVNULL)
    base=read(['git','-C',str(ROOT),'rev-parse','origin/main'])
    run(['docker','exec',WORKER,'git','-C','/work/edict','update-ref','refs/remotes/origin/main',base])
    with (SCRATCH/'source.tar').open('rb') as source_tar:
        run(['docker','exec','-i',WORKER,'tar','xf','-','-C','/work/edict'],stdin=source_tar)
    run(['docker','exec',WORKER,'rm','/work/source.bundle'])
    (SCRATCH/'source.tar').unlink(); (SCRATCH/'source.bundle').unlink()

process=None
try:
    prepare()
    log=SCRATCH/(sys.argv[1]+'.log')
    with log.open('wb') as out:
        process=subprocess.Popen(['docker','exec','--env','RUSTUP_HOME=/cache/rustup','--workdir','/work/edict',WORKER,*sys.argv[2:]],stdout=out,stderr=subprocess.STDOUT,start_new_session=True)
        began=time.monotonic()
        while process.poll() is None:
            guard(metrics())
            if time.monotonic()-began > 2400: raise RuntimeError('workload timeout')
            time.sleep(2)
        code=process.returncode
    guard(metrics())
    if code == 0 and sys.argv[1].endswith('-goldens'):
        exported = SCRATCH/'generated-fixtures.tar'
        with exported.open('wb') as archive:
            run(['docker','exec',WORKER,'tar','cf','-','-C','/work/edict','fixtures'],stdout=archive)
        guard(metrics())
        with tarfile.open(exported) as archive:
            for member in archive.getmembers():
                assert member.name.startswith('fixtures/') or member.name == 'fixtures'
                target=(ROOT/member.name).resolve()
                assert ROOT.resolve() in target.parents
                if member.isdir(): target.mkdir(exist_ok=True,parents=True)
                elif member.isfile():
                    target.parent.mkdir(exist_ok=True,parents=True)
                    with archive.extractfile(member) as source, target.open('wb') as dest:
                        shutil.copyfileobj(source,dest)
                else: raise RuntimeError('unexpected generated archive entry')
        exported.unlink()
    print(json.dumps({'exit_code':code,'log':str(log),'measurements':metrics()}),flush=True)
    tail=log.read_text(errors='replace').splitlines()[-45:]
    print('\n'.join(tail),flush=True)
    sys.exit(code)
except BaseException as e:
    if isinstance(e, SystemExit):
        stop()
        raise
    print(f'GUARD STOP: {type(e).__name__}: {e}',flush=True)
    stop()
    if process is not None and process.poll() is None:
        os.killpg(process.pid,signal.SIGTERM)
        process.wait(timeout=10)
    raise
